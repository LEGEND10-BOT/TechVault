import { spawn } from "node:child_process";
import { setTimeout as sleep } from "node:timers/promises";

const exePath = process.argv[2];
const cdpPort = Number(process.argv[3] ?? 9401);
async function getJson(url, timeoutMs = 800) {
	const ctrl = new AbortController();
	const timer = setTimeout(() => ctrl.abort(), timeoutMs);
	try { const res = await fetch(url, { signal: ctrl.signal }); if (!res.ok) throw new Error(`${res.status}`); return await res.json(); } finally { clearTimeout(timer); }
}
const child = spawn(exePath, [], { env: { ...process.env, WEBVIEW2_ADDITIONAL_BROWSER_ARGUMENTS: `--remote-debugging-port=${cdpPort}` }, stdio: "ignore" });
let ws;
try {
	let page = null;
	for (let i = 0; i < 120 && !page; i++) {
		await sleep(250);
		try { const t = await getJson(`http://127.0.0.1:${cdpPort}/json`); page = t.find((x) => x.type === "page" && (x.url ?? "").startsWith("http://tauri.localhost")); } catch {}
	}
	if (!page) throw new Error("no page target");
	ws = new WebSocket(page.webSocketDebuggerUrl);
	await new Promise((res, rej) => { ws.addEventListener("open", res, { once: true }); ws.addEventListener("error", rej, { once: true }); });
	let id = 0; const pending = new Map();
	ws.addEventListener("message", (ev) => { const m = JSON.parse(ev.data); if (m.id && pending.has(m.id)) { pending.get(m.id)(m); pending.delete(m.id); } });
	const send = (method, params = {}) => new Promise((res) => { const i = ++id; pending.set(i, res); ws.send(JSON.stringify({ id: i, method, params })); });
	const evalJs = async (expr) => (await (await send("Runtime.evaluate", { expression: expr, awaitPromise: true, returnByValue: true })).result)?.result?.value;
	await send("Runtime.enable");
	for (let i = 0; i < 100; i++) { if (await evalJs(`!!document.querySelector('h1')`)) break; await sleep(100); }
	await sleep(1500);
	const fp = await evalJs(`JSON.stringify({
		h1: document.querySelector('h1')?.textContent ?? null,
		badges: Array.from(document.querySelectorAll('span.rounded-card')).map(e => e.textContent.trim()),
		hasMainWorkspace: !!document.getElementById('main-workspace'),
		hasSidebar: !!document.querySelector('aside'),
		appInfoHasDataDir: (window.__lastAppInfo ?? null) !== null ? 'unknown' : 'unknown'
	})`);
	console.log("fingerprint:", fp);
	const ipc = await evalJs(`window.__TAURI_INTERNALS__ ? window.__TAURI_INTERNALS__.invoke('app_info').then(v => JSON.stringify({keys: Object.keys(v.data), profile: v.data.build_profile})).catch(e => 'ERR:' + String(e)) : 'NO_TAURI'`);
	console.log("app_info:", ipc);
} finally {
	try { ws?.close(); } catch {}
	try { child.kill(); } catch {}
}
